# Emacs AI 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `~/.emacs.d` 中落地设计规格（`docs/superpowers/specs/2026-08-31-emacs-ai-design.md`）：gptel 多后端 + gptel-agent 会话 + AGENTS.md 注入 + 工作流命令 + minuet 内联补全 + code-reviewer 子代理。

**Architecture:** 配置源 `init-v2.org`（org-babel tangle）；AI 逻辑在 `site-lisp/omy-ai.el`（纯 elisp 胶水），包声明/后端/键位在 `lisp/init-ai.el`，子代理为 `agents/*.md` 数据文件。全部走 OpenAI 兼容协议。

**Tech Stack:** Emacs 29.2、use-package + package.el（清华 MELPA 镜像已配）、gptel、gptel-agent、minuet、general（leader 键）、projectile、ERT。

## Global Constraints

- 一切配置改动只写在 `init-v2.org` 内，改完必须 tangle；NEVER 手改 tangle 生成的 `.el`
- 全局 `use-package-always-ensure nil` → 本计划每个新包必须显式 `:ensure t`
- site-lisp 不在 load-path（事实核实：grep 无结果）→ 由 init-ai.el 顶部加入
- 项目根判定用 projectile（本仓库既有约定），不可用时回退 `default-directory`
- API key 只从环境变量读：`ZHIPUAI_API_KEY`、`DEEPSEEK_API_KEY`
- 用户命令一律 `omy-ai-` 前缀；leader 菜单用 init-leader-key.el 的 `+general-global-menu!` 宏（SPC 前缀）
- 写类工具免确认仅限 git 仓库内（`gptel-confirm-tool-calls` 置 `nil`，buffer-local）；否则保持 gptel-agent 默认逐次确认
- 分支 `ai-agent`；commit message 小写短语，匹配仓库现状（如 "add gptel backends"）
- 每个任务收尾运行一次回归：`emacs --batch -l ~/.emacs.d/init.el --eval '(princ "AI-LOAD-OK\n")'`，期望输出 `AI-LOAD-OK`。若全量 init 因既有无关模块在 batch 下报错，改用 `emacs --batch -l ~/.emacs.d/lisp/init-package.el -l ~/.emacs.d/lisp/init-ai.el --eval '(princ "AI-LOAD-OK\n")'` 并在 PR 描述中注明
- 首次 batch 加载会经 MELPA 安装 gptel（可能数分钟，属预期）

;;; 会话管理 -------------------------------------------------------
(defvar omy-ai-sessions-dir
  (expand-file-name "omy-ai/sessions/"
                    (or (getenv "XDG_DATA_HOME") "~/.local/share"))
  "gptel 会话文件根目录，按项目 slug 分目录。")

(defun omy-ai--project-root ()
  "当前项目根；projectile 不可用或不在项目内时回退 default-directory。"
  (if (and (fboundp 'projectile-project-p) (projectile-project-p))
      (projectile-project-root)
    default-directory))

(defun omy-ai--project-slug ()
  (omy-ai--slug (file-name-nondirectory
                 (directory-file-name (omy-ai--project-root)))))

(defun omy-ai--slug (name)
  "项目名转文件系统安全 slug，保留中文。"
  (let ((s (downcase name)))
    (setq s (replace-regexp-in-string "[^a-z0-9一-鿿]+" "-" s))
    (setq s (replace-regexp-in-string "\\`-+\\|-+$" "" s))
    (if (string-empty-p s) "default" s)))

(defun omy-ai--session-file (title)
  "当前项目新会话的 org 文件路径（目录不存在则创建）。"
  (let* ((dir (expand-file-name (omy-ai--project-slug) omy-ai-sessions-dir))
         (safe (omy-ai--slug title))
         (stamp (format-time-string "%Y%m%d-%H%M%S")))
    (make-directory dir t)
    (expand-file-name (format "%s-%s.org" stamp safe) dir)))

(defun omy-ai-session-new (&optional title)
  "为当前项目新建 gptel 会话文件并打开。"
  (interactive "s会话标题 (回车用默认): ")
  (let* ((title (if (or (not title) (string-empty-p title)) "session" title))
         (path (omy-ai--session-file title)))
    (find-file path)
    (gptel-mode 1)
    (insert (format "#+title: %s\n\n" title))))

(defun omy-ai-session-open ()
  "列出并打开当前项目的既有 gptel 会话（最新在前）。"
  (interactive)
  (let* ((dir (expand-file-name (omy-ai--project-slug) omy-ai-sessions-dir))
         (files (when (file-directory-p dir)
                  (sort (directory-files dir t "\\.org\\'")
                        #'file-newer-than-file-p))))
    (if (not files)
        (user-error "omy-ai: 项目还没有会话，先用 SPC a s 新建")
      (let* ((names (mapcar #'file-name-nondirectory files))
             (pick (completing-read "打开会话: " names nil t)))
        (find-file (expand-file-name pick dir))
        (gptel-mode 1)))))
```

;;; AGENTS.md 注入与 agent 会话 ------------------------------------
(defun omy-ai--git (args)
  "在项目根执行 git ARGS，返回 (成功? . 输出)。"
  (let ((default-directory (omy-ai--project-root))
        (out "") status)
    (setq out (with-output-to-string
                (with-current-buffer standard-output
                  (setq status (apply #'call-process "git" nil (list t nil) nil args)))))
    (cons (zerop status) out)))

(defun omy-ai--inside-git-p (&optional dir)
  "DIR 是否位于 git 工作树内。"
  (let ((default-directory (or dir (omy-ai--project-root))))
    (and (executable-find "git")
         (car (omy-ai--git '("rev-parse" "--is-inside-work-tree"))))))

(defun omy-ai--rules-path (&optional root)
  "项目根 AGENTS.md 路径，不存在时 nil。"
  (let ((p (expand-file-name "AGENTS.md" (or root (omy-ai--project-root)))))
    (and (file-exists-p p) p)))

(defun omy-ai-agent ()
  "为当前项目打开 gptel-agent 会话。
git 仓库内：写类工具免确认（直写 + git 兜底），并注入 AGENTS.md。
仓库外：保持 gptel-agent 默认逐次确认，仅提示。"
  (interactive)
  (let* ((root (omy-ai--project-root))
         (in-git (omy-ai--inside-git-p root))
         (rules (omy-ai--rules-path root)))
    (unless in-git
      (message "omy-ai: 不在 git 仓库内，写类工具保持逐次确认"))
    (let ((default-directory root))
      (gptel-agent))
    (let ((buf (or (and (bound-and-true-p gptel-mode) (current-buffer))
                   (seq-find (lambda (b) (buffer-local-value 'gptel-mode b))
                             (buffer-list)))))
      (when buf
        (with-current-buffer buf
          (when in-git (setq-local gptel-confirm-tool-calls nil))
          (when rules (gptel-add-file rules)))))))
```

```elisp
;;; 工作流命令 ------------------------------------------------------
(defvar omy-ai-commit-system
  "你是 commit message 生成器。基于 diff 写一条简洁的 git commit
message：第一行 <=50 字符概括，正文可选说明动机，用 diff 的主要语言。
只输出 message 本身，不要 markdown 代码块。")

(defvar omy-ai-review-system
  "你是资深代码审查者。只报告真实问题（正确性、边界、并发、安全、
测试缺口），每条给出文件/位置与理由，按严重程度排序；没有问题就明确
说 LGTM。用中文。")

(defun omy-ai--staged-diff ()
  "staged diff；不在 git 仓库或无暂存时 user-error。"
  (pcase-let ((`(,ok . ,out) (omy-ai--git '("diff" "--cached"))))
    (cond ((not ok) (user-error "omy-ai: 当前不在 git 仓库中"))
          ((string-empty-p (string-trim out))
           (user-error "omy-ai: 没有 staged 改动（先 git add）"))
          (t (string-trim out)))))

(defun omy-ai--show (buffer-name response)
  "在 BUFFER-NAME 中显示 LLM 响应。"
  (with-current-buffer (get-buffer-create buffer-name)
    (let ((inhibit-read-only t))
      (erase-buffer)
      (insert response))
    (delay-mode-hooks (org-mode))
    (display-buffer (current-buffer))))

(defun omy-ai--lang ()
  "当前 major-mode 对应的语言名，如 python。"

;;; 手动 compaction --------------------------------------------------
(defun omy-ai-compact ()
  "把当前会话交给模型压缩为交接摘要，重置为新起点。
gptel 无自动 compaction，长会话手动执行本命令。"
  (interactive)
  (unless (bound-and-true-p gptel-mode)
    (user-error "omy-ai: 当前不在 gptel 会话中"))
  (let ((buf (current-buffer))
        (history (buffer-substring-no-properties (point-min) (point-max))))
    (message "omy-ai: 压缩会话 ...")
    (gptel-request
        (format "把下面的 agent 会话记录压缩成简洁的交接摘要：目标、
已做决定、当前状态、下一步。保留关键文件路径与命令。\n\n%s" history)
      :system "你会压缩 agent 会话记录：保留决策与状态，删除冗余过程。用中文。"
      :callback (lambda (response _info)
                  (with-current-buffer buf
                    (let ((inhibit-read-only t))
                      (erase-buffer)
                      (insert (format "#+title: compacted %s\n\n%s\n"
                                      (format-time-string "%F %T") response)))
                    (goto-char (point-max))
                    (message "omy-ai: 会话已压缩"))))))
```
